<script>
const coin = document.querySelector('.mexa-coin');
const piggybank = document.getElementById('mexa-piggybank');

// وقتی کشیده میشه
coin.addEventListener('dragstart', (e) => {
  e.dataTransfer.setData('coin', 'dragging');
});

// وقتی میره روی قلک
piggybank.addEventListener('dragover', (e) => {
  e.preventDefault();
  piggybank.classList.add('over');
});

// وقتی میره بیرون
piggybank.addEventListener('dragleave', () => {
  piggybank.classList.remove('over');
});

// وقتی بندازن داخل قلک
piggybank.addEventListener('drop', () => {
  piggybank.classList.remove('over');
  // 👉 انتقال به صفحه پرداخت
  window.location.href = "onlinedonation.html"; 
});
</script>
